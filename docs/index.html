context.yamlVisa källa 
context:
  system:
    id: example
    name: Example System
  parties:
    - id: user
      type: person
      name: User
  flows:
    - id: user-input
      from: user
      to: example
      name: User input
      initiative: user
    - id: system-information
      from: example
      to: user
      name: System information

    - id: process-input
      name: Process input

      display: "01"
      name: Input submitted
  flows:
    - trigger: User submits input
      pulse: input-submitted
      to: process-input

deployment.yamlVisa källa 
deployment:
  environment:
    file: .env
    variables:
      - name: HTTP_ADDR
        required: false
        default: 127.0.0.1:8080
  hosts:
    - id: application-host
      name: Application host
      type: machine
      os: linux
      architecture: amd64
  programs:
    - id: example-application
      name: Example application
      host: application-host
      type: os-process
      role: server-with-web-ui
      implementation:
        language: go
        platform: server-rendered-html
      command: ./example-application
      environment:
        file: .env
      ports:
        - id: web
          name: Web UI
          port: 8080
          application: http
          exposure: host
      health-check:
        type: http
        path: /health
      restart: on-failure
      resources:
        memory: 128MiB
    - id: data-stack
      name: Data stack
      host: application-host
      type: docker-compose
      environment:
        file: .env
      services:
        - id: database
          name: Database
          role: database
          implementation:
            platform: postgres
          ports:
            - id: postgres
              name: PostgreSQL
              port: 5432
              application: postgres
              exposure: internal
          volumes:
            - id: data
              source: database-data
              target: /var/lib/postgresql/data
          health-check:
            type: command
            command: pg_isready
          restart: unless-stopped
  connections:
    - id: application-database
      name: Store application data
      from: example-application
      to: data-stack.database.postgres
      transport: tcp
      application: postgres
requirements.yamlVisa källa 
requirements:
  context.flow.user-input:
    - The system shall accept user input.
  pulse.behavior.process-input:
    - The system shall process submitted input.
  ui.action.main.submit-input:
    - The user shall be able to submit input.
  deployment.host.application-host:
    - The application shall run on a Linux host.
  deployment.program.example-application:
    - The server and Web UI shall be delivered as one operating-system process.
  deployment.program.example-application.port.web:
    - The Web UI shall listen on port 8080.
  deployment.service.data-stack.database:
    - Persistent application data shall be stored in PostgreSQL.
  deployment.connection.application-database:
    - The application shall connect to PostgreSQL over the internal deployment network.
